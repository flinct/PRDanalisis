    id: standard-multi-agent-v1

      - qa_pre

      - qa_post
      - reviewer_gate_c
    gates:

        reviewer_role: reviewer

      gate_c:
        reviewer_role: reviewer
        input_roles: [coder_automation, qa_agent]

      starts_after: reviewer_gate_b

        reviewer_role: reviewer
        input_roles: [coder_automation, qa_agent]
    freeze:
      enabled: true
      starts_after: reviewer_gate_b
